## ============================================================
## Uresh-Vivek Precision (2nd edition)
## Transcribed from bidlab/input/Uresh-Vivek Precision2.pdf, for use
## with the Bidder rule language (see shared/hand-spec.md).
##
## SCOPE: this file covers the uncontested partnership auction --
## opening bids and the responses/rebids that follow when the
## opponents pass throughout. As with bws_input.txt, competitive
## material is omitted by request, and because this rule language has
## no token for an opposing call in a "$." sequence anyway:
##   - Ch 2.3 (Dealing with Intervention, after 1N)
##   - Ch 7 (Dealing with intervention over 1C)
##   - Ch 14 / 14.1 (When Opponents Open / Preempt)
##   - Ch 15 (Competitive Auctions: support doubles, NFB, responsive
##     doubles, maximal doubles, cue bids in defense, etc.)
##   - Ch 16 (Competitive Lebensohl)
## Ch 17 (Bidding by a Passed Hand) describes alternate meanings for a
## hand that has already passed; this language has no notion of
## "passed hand" as a distinct hand property, so it isn't separately
## encoded either.
## Ch 18/19 (change log, open issues) are meta-commentary, not bids.
## Ch 20 (Discontinued Sequences) is explicitly superseded by Ch 5/6
## in the source document and is skipped entirely for that reason --
## including it would silently contradict the current-system rules
## given earlier in the same chapters.
##
## COLOR-CODING IN THE SOURCE: black = normal current text, red =
## changes to the original system (both treated as current/binding
## here), blue = unusual treatments worth a comment. Purple ("proposed
## changes, not discussed or agreed") is not part of the agreed
## system and is omitted -- it only occurs in the competitive
## material anyway.
##
## DEPTH: this is an unusually deep relay-based system (Control-Asking
## Bids, Trump-Asking Bids, Special Trump-Asking Bids, Epsilon Bids,
## the Queen/Jack Asking Relay, the Telescope Ask, the 4CD convention,
## and multi-level "zoom" distributional relays after 1C - 2C/2D/2H).
## These are generic step-response protocols layered on top of
## whatever auction established them; exhaustively enumerating every
## auction they can attach to (the source's own Ch 20.3-20.7 alone
## runs to pages of tables) would run to thousands of lines. They are
## implemented here at the positions explicitly named in the source
## -- the clearest, most central instances -- and otherwise documented
## as named conventions: the general step-reply pattern is given once,
## near its first use, rather than repeated at every attachment point.
##
## ARITHMETIC LIMITATION (as in bws_input.txt): the rule language has
## no addition operator over card-presence booleans, so a true Double
## Key-Card (both-suits-at-once) response, and 4CD-style multi-strain
## asks beyond the concrete cases spelled out in the source, can't be
## built from the available primitives; only single-suit KCB/Kickback
## is fully modeled.
## ============================================================


## ============================================================
## HELPERS
## ============================================================

$Any := Points >= 0 ;

$balanced := shape [4,3,3,3] OR shape [4,4,3,2] OR shape [5,3,3,2] ;
## 5-4-2-2 / 6-3-2-2 only count as notrump shape here when the long
## suit is a minor (Ch 2 requirement); "if 6-carded it should be weak"
## is a judgment call with no HCP-based representation and is not
## separately modeled.
$semiBalMinor := (shape [5,4,2,2] AND ((Clubs ?= 5) OR (Diamonds ?= 5))) OR
                 (shape [6,3,2,2] AND ((Clubs ?= 6) OR (Diamonds ?= 6))) ;
$ntShape := $balanced OR $semiBalMinor ;
$noMajor4 := (Spades < 4) AND (Hearts < 4) ;
$noMajor5 := (Spades < 5) AND (Hearts < 5) ;

## ---- Ace-count helpers (Gerber / plain Blackwood; no native "how
## many aces" keyword). ----
$aces0 := NOT Sa AND NOT Ha AND NOT Da AND NOT Ca ;
$aces1 := (Sa AND NOT Ha AND NOT Da AND NOT Ca) OR
          (NOT Sa AND Ha AND NOT Da AND NOT Ca) OR
          (NOT Sa AND NOT Ha AND Da AND NOT Ca) OR
          (NOT Sa AND NOT Ha AND NOT Da AND Ca) ;
$aces2 := (Sa AND Ha AND NOT Da AND NOT Ca) OR
          (Sa AND NOT Ha AND Da AND NOT Ca) OR
          (Sa AND NOT Ha AND NOT Da AND Ca) OR
          (NOT Sa AND Ha AND Da AND NOT Ca) OR
          (NOT Sa AND Ha AND NOT Da AND Ca) OR
          (NOT Sa AND NOT Ha AND Da AND Ca) ;
$aces3 := (NOT Sa AND Ha AND Da AND Ca) OR
          (Sa AND NOT Ha AND Da AND Ca) OR
          (Sa AND Ha AND NOT Da AND Ca) OR
          (Sa AND Ha AND Da AND NOT Ca) ;
$aces4 := Sa AND Ha AND Da AND Ca ;

## ---- Control-count step helpers (A = 2, K = 1) for Control-Asking
## Bids (Ch 4). ----
$ctl0 := Controls ?= 0 ;
$ctl1 := Controls ?= 1 ;
$ctl2 := Controls ?= 2 ;
$ctl3 := Controls ?= 3 ;
$ctl4 := Controls ?= 4 ;
$ctl5 := Controls ?= 5 ;
$ctl6 := Controls >= 6 ;

$gf := Points >= 13 ;
$inv := 10 TO 12 Points ;
$slamish := (Points >= 12) AND (Controls >= 4) ;


## ============================================================
## Ch 1: OPENING BIDS
## ============================================================

## One club: 17+ HCP, any shape. All other openings deny 17 HCP.
$.1C. := Points >= 17 ;

## Three notrump: gambling, solid seven-card minor, no outside A/K
## (approximated as in bws_input.txt via suit points, since seat/
## position-dependent nuance has no representation here).
$soundClubSuit := (Clubs >= 7) AND (Cpts >= 6) ;
$soundDiamondSuit := (Diamonds >= 7) AND (Dpts >= 6) ;
$.3N. := ($soundClubSuit OR $soundDiamondSuit) AND (Points <= 10) AND NOT $.1C. ;

## Two hearts: 11-16, 4-4-1-4 or 4-4-0-5 (exact spades-hearts-
## diamonds-clubs order per the glossary's hyphenated notation); or
## 14-16, 4-3-1-5/3-4-1-5 with QTx-or-better in the three-card suit.
$.2H. := NOT ($.1C. OR $.3N.) AND
         (((11 TO 16 Points) AND (shape [4-4-1-4] OR shape [4-4-0-5])) OR
          ((14 TO 16 Points) AND (shape [4-3-1-5]) AND Hq AND Ht) OR
          ((14 TO 16 Points) AND (shape [3-4-1-5]) AND Sq AND St)) ;

## Two clubs: 11-16, five clubs and a 4-card major, or 6+ clubs, or
## 15-16 with 5+ clubs and 4 diamonds.
$.2C. := NOT ($.1C. OR $.3N. OR $.2H.) AND
         (((11 TO 16 Points) AND (Clubs ?= 5) AND ((Spades >= 4) OR (Hearts >= 4))) OR
          ((11 TO 16 Points) AND (Clubs >= 6)) OR
          ((15 TO 16 Points) AND (Clubs >= 5) AND (Diamonds ?= 4))) ;

## One notrump: 12-14 balanced/semi-balanced; if exactly 14, no
## 5-card suit (so 5-3-3-2/5-4-2-2/6-3-2-2 are excluded at 14). The
## text allows as few as 11 HCP non-vulnerable; not separately
## representable here (no vulnerability keyword), so the floor is set
## at 11 uniformly.
$.1N. := NOT ($.1C. OR $.3N. OR $.2H. OR $.2C.) AND
         (((11 TO 13 Points) AND $ntShape) OR
          ((Points ?= 14) AND (shape [4,3,3,3] OR shape [4,4,3,2]))) ;

## Two diamonds: weak two in either major (opener knows which; not
## separately distinguishable as two different hand-typed rules since
## the call itself is the same "$.2D." node either way).
$weakTwoStrength := 5 TO 10 Points ;
$.2D. := $weakTwoStrength AND
         (((Hearts ?= 6) AND (Spades <= 3) AND (Diamonds <= 4) AND (Clubs <= 4)) OR
          ((Spades ?= 6) AND (Hearts <= 3) AND (Diamonds <= 4) AND (Clubs <= 4))) AND
         NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N.) ;

## Two spades: weak, 5-5+ in the reds or in the blacks.
$.2S. := $weakTwoStrength AND
         ((Hearts >= 5 AND Diamonds >= 5) OR (Spades >= 5 AND Clubs >= 5)) AND
         NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N. OR $.2D.) ;

## Two notrump: weak, 5-5+ in the majors or in the minors.
$.2N. := $weakTwoStrength AND
         ((Hearts >= 5 AND Spades >= 5) OR (Clubs >= 5 AND Diamonds >= 5)) AND
         NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N. OR $.2D. OR $.2S.) ;

## Three-level preempts (natural), using the Maj/Min macros for the
## symmetric case.
$.3Maj. := (5 TO 10 Points) AND (Maj >= 7) AND
           NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N. OR $.2D. OR $.2S. OR $.2N.) ;
$.3Min. := (5 TO 10 Points) AND (Min >= 7) AND
           NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N. OR $.2D. OR $.2S. OR $.2N.) ;

## One-level suit openings. Standard priority for a two-suited major
## hand: bid the longer; with equal length, spades.
$majorPrioritySpade := (Spades >= 5) AND (Spades >= Hearts) ;
$majorPriorityHeart := (Hearts >= 5) AND (Hearts > Spades) ;
$openingRange := 11 TO 16 Points ;

$.1S. := $openingRange AND $majorPrioritySpade AND NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N.) ;
$.1H. := $openingRange AND $majorPriorityHeart AND
         NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N. OR $.1S.) ;

## One diamond: 11-16 with 4+ diamonds, or balanced 14-16 with no
## qualifying major (the residual balanced-15-16 hand that is too
## strong for 1NT and does not match two clubs' specific shape).
$.1D. := $openingRange AND (Spades < 5) AND (Hearts < 5) AND
         ((Diamonds >= 4) OR $balanced) AND
         NOT ($.1C. OR $.3N. OR $.2H. OR $.2C. OR $.1N.) ;

## Four of a minor: NAMYATS. 4C transfers to an independent 7+ heart
## suit, 4D to an independent 7+ spade suit, 8-9 playing tricks
## (approximated here by suit points, since "playing tricks" is not a
## native measure).
$.4C. := (Hearts >= 7) AND (Hpts >= 7) AND NOT $.1C. ;
$.4D. := (Spades >= 7) AND (Spts >= 7) AND NOT $.1C. ;

## Four of a major: to play, weaker than the NAMYATS 4m.
$.4H. := (Hearts >= 7) AND (Points <= 9) AND (Hpts < 7) AND NOT $.1C. ;
$.4S. := (Spades >= 7) AND (Points <= 9) AND (Spts < 7) AND NOT $.1C. ;

## Pass.
$.P. := NOT ($.1C. OR $.1D. OR $.1H. OR $.1S. OR $.1N. OR
             $.2C. OR $.2D. OR $.2H. OR $.2S. OR $.2N. OR
             $.3C. OR $.3D. OR $.3H. OR $.3S. OR $.3N. OR
             $.4C. OR $.4D. OR $.4H. OR $.4S.) ;


## ============================================================
## Ch 2 / 2.1 / 2.2: THE ONE-NOTRUMP OPENING
## ============================================================

## Two spades: 5-4-or-better in the minors, either preemptive or a
## strong/slammish hand.
$bothMinors54 := (Clubs >= 5 AND Diamonds >= 4) OR (Diamonds >= 5 AND Clubs >= 4) ;
$.1N.2S. := $noMajor4 AND $bothMinors54 AND ((Points <= 7) OR $slamish) ;

## Two clubs: Puppet Stayman -- one or two 4-card majors (inv+), an
## invitational-or-better long minor, or a balanced invitational hand
## with or without a major.
$.1N.2C. := (Points >= 10) AND NOT $bothMinors54 AND
            ((Spades ?= 4) OR (Hearts ?= 4) OR
             ($noMajor4 AND $balanced) OR
             ($noMajor4 AND (Clubs >= 5 OR Diamonds >= 5))) ;

## Jacoby transfers (denies 4+ of the other major unless very weak).
$.1N.2D. := (Hearts >= 5) AND ((Spades < 4) OR (Points <= 6)) ;
$.1N.2H. := (Spades >= 5) AND ((Hearts < 4) OR (Points <= 6)) ;

## Unconditional transfers to clubs/diamonds: signoff or game force
## (invitational hands with a long minor go via Puppet Stayman above).
$.1N.2N. := (Clubs >= 5) AND ((Points <= 9) OR (Points >= 13)) ;
$.1N.3C. := (Diamonds >= 5) AND ((Points <= 9) OR (Points >= 13)) ;

## Three diamonds: minor-suit Stayman, strong 2-3-4-4 shape (either
## major split).
$.1N.3D. := (Clubs ?= 4) AND (Diamonds ?= 4) AND
            (((Spades ?= 2) AND (Hearts ?= 3)) OR ((Spades ?= 3) AND (Hearts ?= 2))) AND
            (Points >= 10) ;
$.1N.3D.3H. := Clubs ?= 4 ;
$.1N.3D.3S. := Diamonds ?= 4 ;
$.1N.3D.3N. := $noMajor4 ;

## Three of a major: 3-suited hand, short in the bid major, 4 cards in
## the other, minors split 5-3/4-4/5-4.
$minorSplit3M := (Clubs ?= 5 AND Diamonds ?= 3) OR (Clubs ?= 3 AND Diamonds ?= 5) OR
                 (Clubs ?= 4 AND Diamonds ?= 4) OR (Clubs ?= 5 AND Diamonds ?= 4) OR
                 (Clubs ?= 4 AND Diamonds ?= 5) ;
$.1N.3H. := (Hearts <= 1) AND (Spades ?= 4) AND $minorSplit3M AND (Points >= 10) ;
$.1N.3S. := (Spades <= 1) AND (Hearts ?= 4) AND $minorSplit3M AND (Points >= 10) ;

## Three/four notrump: natural.
$.1N.3N. := $balanced AND $noMajor4 AND (11 TO 15 Points) ;
$.1N.4N. := $balanced AND $noMajor4 AND (16 TO 17 Points) ;

## Gerber.
$.1N.4C. := $slamish AND $noMajor4 ;
$.1N.4C.4D. := $aces0 ;
$.1N.4C.4H. := $aces1 ;
$.1N.4C.4S. := $aces2 ;
$.1N.4C.4N. := $aces3 ;

## Texas transfers; 4S over 4H, or 4N over 4S, is Kickback (the bid
## above the trump suit asks for key cards). Step order in this system
## is 1/4, 0/3, 2-without-trump-Q, 2-with-trump-Q (Ch 4.2/4.3) --
## different from bws_input.txt's BWS-style 0/3, 1/4, 2, 2+Q ordering.
$.1N.4D. := (Hearts >= 6) AND (8 TO 11 Points) ;
$.1N.4H. := (Spades >= 6) AND (8 TO 11 Points) ;
$.1N.4D.4H. := Points >= 0 ;
$.1N.4H.4S. := Points >= 0 ;
$.1N.4D.4H.4S. := $slamish ;
$.1N.4D.4H.4S.4N. := (Hkcs ?= 1) OR (Hkcs ?= 4) ;
$.1N.4D.4H.4S.5C. := (Hkcs ?= 0) OR (Hkcs ?= 3) ;
$.1N.4D.4H.4S.5D. := (Hkcs ?= 2) AND NOT Hq ;
$.1N.4D.4H.4S.5H. := (Hkcs ?= 2) AND Hq ;
$.1N.4H.4S.4N. := $slamish ;
$.1N.4H.4S.4N.5C. := (Skcs ?= 1) OR (Skcs ?= 4) ;
$.1N.4H.4S.4N.5D. := (Skcs ?= 0) OR (Skcs ?= 3) ;
$.1N.4H.4S.4N.5H. := (Skcs ?= 2) AND NOT Sq ;
$.1N.4H.4S.4N.5S. := (Skcs ?= 2) AND Sq ;

## ---- Ch 2.1: Bidding After Puppet Stayman ----

## Opener bids a 5-card major if he has one, else two diamonds.
$.1N.2C.2H. := Hearts >= 5 ;
$.1N.2C.2S. := (Spades >= 5) AND (Hearts < 5) ;
$.1N.2C.2D. := (Hearts < 5) AND (Spades < 5) ;

## Over 1N-2C-2H, two spades agrees hearts and starts a Telescope Ask;
## over 1N-2C-2S, three hearts is a general slam try inviting cue
## bids. 1N-2C-2M-4N is quantitative; 1N-2C-2H-4S is Kickback.
$.1N.2C.2H.2S. := (Hearts >= 3) AND $slamish ;
$.1N.2C.2S.3H. := (Spades >= 3) AND $slamish ;
$.1N.2C.2H.4N. := (16 TO 17 Points) AND $balanced ;
$.1N.2C.2S.4N. := (16 TO 17 Points) AND $balanced ;
$.1N.2C.2H.4S. := $slamish ;
$.1N.2C.2H.4S.4N. := (Hkcs ?= 1) OR (Hkcs ?= 4) ;
$.1N.2C.2H.4S.5C. := (Hkcs ?= 0) OR (Hkcs ?= 3) ;
$.1N.2C.2H.4S.5D. := (Hkcs ?= 2) AND NOT Hq ;
$.1N.2C.2H.4S.5H. := (Hkcs ?= 2) AND Hq ;

## Responder's rebids over 1N-2C-2D.
$.1N.2C.2D.2H. := (Hearts ?= 4) AND (Points >= 10) ;
$.1N.2C.2D.2S. := (Spades ?= 4) AND (Hearts < 4) AND (Points >= 10) ;
$.1N.2C.2D.2N. := $noMajor4 AND $inv ;
$.1N.2C.2D.3C. := (Clubs >= 5) AND $noMajor4 AND $inv ;
$.1N.2C.2D.3D. := (Diamonds >= 5) AND $noMajor4 AND $inv ;
$.1N.2C.2D.3H. := (Hearts >= 5) AND (Spades >= 5) AND $inv ;
$.1N.2C.2D.3S. := (Hearts ?= 5) AND (Spades ?= 5) AND $gf AND ((Clubs <= 1) OR (Diamonds <= 1)) ;
$.1N.2C.2D.3N. := $balanced AND $noMajor4 AND (13 TO 15 Points) ;
$.1N.2C.2D.4C. := (Hearts ?= 5) AND (Spades ?= 5) AND (Clubs ?= 0) AND $slamish ;
$.1N.2C.2D.4D. := (Hearts ?= 5) AND (Spades ?= 5) AND (Diamonds ?= 0) AND $slamish ;

## Opener's continuations over the 5-5-forcing three spades.
$.1N.2C.2D.3S.4C. := (Hearts >= 3) AND (Points >= 14) ;
$.1N.2C.2D.3S.4D. := (Spades >= 3) AND (Points >= 14) ;
$.1N.2C.2D.3S.3N. := (Hearts >= 3) AND (Spades >= 3) AND (14 TO 16 Points) ;
$.1N.2C.2D.3S.4H. := (Hearts >= 3) AND (Points <= 13) ;

## Opener's rebids after 1N-2C-2D-2H (the one-round-force major ask).
$.1N.2C.2D.2H.2S. := (Spades ?= 4) AND (Hearts < 4) ;
$.1N.2C.2D.2H.2N. := (Hearts < 4) AND (Spades < 4) AND (Points <= 13) ;
$.1N.2C.2D.2H.3C. := (Hearts < 4) AND (Spades < 4) AND (Points >= 14) ;
$.1N.2C.2D.2H.3D. := (Hearts ?= 4) AND (Points <= 13) ;
$.1N.2C.2D.2H.3H. := (Hearts >= 4) AND (Points <= 12) ;
$.1N.2C.2D.2H.3N. := (Hearts >= 4) AND (Points >= 14) AND NOT (Sa OR Da OR Ca) ;
$.1N.2C.2D.2H.4C. := (Hearts >= 4) AND (Points >= 14) AND Ca ;
$.1N.2C.2D.2H.4D. := (Hearts >= 4) AND (Points >= 14) AND Da ;

## Opener's rebids after 1N-2C-2D-2S.
$.1N.2C.2D.2S.2N. := (Spades < 4) AND (Points <= 13) ;
$.1N.2C.2D.2S.3C. := (Spades < 4) AND (Points >= 14) ;
$.1N.2C.2D.2S.3D. := (Spades ?= 4) AND (Points <= 13) ;
$.1N.2C.2D.2S.3S. := (Spades >= 4) AND (Points <= 12) ;
$.1N.2C.2D.2S.3N. := (Spades >= 4) AND (Points >= 14) AND NOT (Ha OR Da OR Ca) ;
$.1N.2C.2D.2S.3H. := (Spades >= 4) AND (Points >= 14) AND Ha ;
$.1N.2C.2D.2S.4C. := (Spades >= 4) AND (Points >= 14) AND Ca ;
$.1N.2C.2D.2S.4D. := (Spades >= 4) AND (Points >= 14) AND Da ;

## Representative continuation once opener has denied a fit (2N/3C):
## three of a minor is invitational with a 5+ suit; three of a major
## shows 5-4 in the majors, 4 in the first-bid suit.
$.1N.2C.2D.2H.2N.3C. := (Clubs >= 5) AND $inv ;
$.1N.2C.2D.2H.2N.3D. := (Diamonds >= 5) AND $inv ;
$.1N.2C.2D.2H.2N.3H. := (Hearts >= 4) AND (Spades >= 5) AND $inv ;
$.1N.2C.2D.2H.2N.3S. := (Hearts >= 4) AND (Spades >= 5) AND $gf ;

## After 1N-2C-2D-2H-2S, three hearts is minor-suit Stayman.
$.1N.2C.2D.2H.2S.3H. := (Hearts < 4) AND (Spades < 4) AND $slamish ;
$.1N.2C.2D.2H.2S.3H.3S. := Diamonds ?= 4 ;
$.1N.2C.2D.2H.2S.3H.3N. := shape [4,3,3,3] ;
$.1N.2C.2D.2H.2S.3H.4C. := (Clubs ?= 4) AND $ctl0 ;
$.1N.2C.2D.2H.2S.3H.4D. := (Clubs ?= 4) AND $ctl1 ;
$.1N.2C.2D.2H.2S.3H.4H. := (Clubs ?= 4) AND $ctl2 ;

## ---- Ch 2.2: Bidding after Jacoby Transfers ----

## Superaccepts: bid a doubleton (spade doubleton over the heart
## transfer), or a long-suit try with an unspecified long suit (next
## step asks), or honors in the doubleton.
$.1N.2D.2S. := (Hearts >= 4) AND (Points >= 14) AND (Spades != 2) ;
$.1N.2D.2N. := (Hearts >= 4) AND (Spades ?= 2) AND (Points >= 14) AND NOT (Sq OR Sk OR Sa) ;
$.1N.2D.3H. := (Hearts >= 4) AND (Spades ?= 2) AND (Points >= 14) AND (Sq OR Sk OR Sa) ;
$.1N.2D.2H. := NOT ((Hearts >= 4) AND (Points >= 14)) ;

$.1N.2H.2N. := (Spades >= 4) AND (Points >= 14) AND NOT ((Hearts ?= 2) AND (Hq OR Hk OR Ha)) ;
$.1N.2H.3S. := (Spades >= 4) AND (Hearts ?= 2) AND (Points >= 14) AND (Hq OR Hk OR Ha) ;
$.1N.2H.2S. := NOT ((Spades >= 4) AND (Points >= 14)) ;

## Responder's rebids after the plain heart accept: new major = trump
## ask/slam try; new minor = natural GF; rebid of suit = natural
## invitational; notrump = natural (no Blackwood); jump to game =
## invitational RKC (immediate key-card ask).
$.1N.2D.2H.2S. := (Hearts >= 5) AND (Spades >= 4) AND $slamish ;
$.1N.2D.2H.3C. := (Hearts >= 5) AND (Clubs >= 5) AND $gf ;
$.1N.2D.2H.3D. := (Hearts >= 5) AND (Diamonds >= 5) AND $gf ;
$.1N.2D.2H.3H. := (Hearts ?= 5) AND $inv ;
$.1N.2D.2H.3N. := (Hearts ?= 5) AND $balanced AND (10 TO 13 Points) ;
$.1N.2D.2H.4H. := (Hearts >= 6) AND $inv ;
$.1N.2D.2H.4H.5C. := (Hkcs ?= 1) OR (Hkcs ?= 4) ;
$.1N.2D.2H.4H.5D. := (Hkcs ?= 0) OR (Hkcs ?= 3) ;
$.1N.2D.2H.4H.5H. := (Hkcs ?= 2) AND NOT Hq ;
$.1N.2D.2H.4H.5S. := (Hkcs ?= 2) AND Hq ;

## Bidding after 1N-2H-2S-3H: general slam try in spades, denies a
## singleton; opener's rebids concern trump quality.
$top2Spades := (Sa AND Sk) OR (Sa AND Sq) OR (Sk AND Sq) ;
$top3Spades := Sa AND Sk AND Sq ;
$.1N.2H.2S.3H. := (Spades >= 4) AND $slamish AND (Hearts >= 2) AND (Diamonds >= 2) AND (Clubs >= 2) ;
$.1N.2H.2S.3H.3S. := (Spades ?= 3) AND NOT ($top2Spades OR $top3Spades) ;
$.1N.2H.2S.3H.3N. := ((Spades ?= 3) AND ($top2Spades OR $top3Spades)) OR
                     ((Spades ?= 4) AND NOT $top2Spades) ;
$.1N.2H.2S.3H.4C. := (Spades >= 4) AND $top2Spades AND (Clubs <= 1) ;
$.1N.2H.2S.3H.4D. := (Spades >= 4) AND $top2Spades AND (Diamonds <= 1) ;

## Bidding after 1N-2D-2H-2S: asks trump length; responses show 2, 3
## and 4 trumps. Step 1 (here) is STA, step 2 is controls/Epsilon.
$.1N.2D.2H.2S.2N. := Hearts ?= 2 ;
$.1N.2D.2H.2S.3C. := Hearts ?= 3 ;
$.1N.2D.2H.2S.3D. := Hearts >= 4 ;


## ============================================================
## Ch 5 / 5.1-5.8, Ch 6: THE ONE-CLUB OPENING
## ============================================================
## NOTATION REMINDER (per the source glossary): a hyphenated pattern
## like "4-4-1-4" is an EXACT spades-hearts-diamonds-clubs shape (the
## dash-delimited shape literal here); a dotted pattern like "4.4.4.1"

## 11-16, 4-4-1-4/4-4-0-5, or 14-16, 4-3-1-5/3-4-1-5 with QTx+
## (already defined in Ch 1).

$.2H.2S. := Points <= 8 ;
$.2H.3C. := (Clubs >= 5) AND (Points <= 8) ;
$.2H.2N. := Points >= 9 ;
$.2H.3D. := (Clubs >= 4) AND ((Spades >= 4) OR (Hearts >= 4)) AND (9 TO 12 Points) ;
$.2H.3H. := (Hearts >= 3) AND $gf ;
$.2H.3S. := (Spades >= 3) AND $gf ;
$.2H.4C. := (Clubs >= 3) AND $gf ;

## ---- Bidding after 2H-2N ----
$.2H.2N.3C. := shape [3-4-1-5] ;
$.2H.2N.3D. := shape [4-3-1-5] ;
$.2H.2N.3H. := shape [4-4-1-4] AND (11 TO 13 Points) ;
$.2H.2N.3S. := shape [4-4-1-4] AND (14 TO 16 Points) AND NOT (Da OR Dk OR Dq) ;
$.2H.2N.3N. := shape [4-4-1-4] AND (14 TO 16 Points) AND (Da OR Dk OR Dq) ;
$.2H.2N.4C. := shape [4-4-0-5] AND (11 TO 13 Points) ;
$.2H.2N.4D. := shape [4-4-0-5] AND (14 TO 16 Points) AND NOT (Ha OR Hk) ;
$.2H.2N.4H. := shape [4-4-0-5] AND (14 TO 16 Points) AND (Ha OR Hk) ;

## ---- Bidding after 2H-3D ----
$.2H.3D.3H. := (shape [4-4-1-4] AND (11 TO 13 Points)) OR (shape [4-4-0-5] AND (11 TO 12 Points)) ;
$.2H.3D.3S. := shape [4-3-1-5] ;
$.2H.3D.3S.3N. := Da OR Dk OR Dq ;
$.2H.3D.3N. := shape [3-4-1-5] AND (Da OR Dk OR Dq) ;
$.2H.3D.4C. := shape [3-4-1-5] AND NOT (Da OR Dk OR Dq) ;
$.2H.3D.4D. := (shape [4-4-1-4] AND (14 TO 16 Points)) OR (shape [4-4-0-5] AND (13 TO 15 Points)) ;
$.2H.3D.4H. := shape [4-4-0-5] AND (15 TO 16 Points) ;


## ============================================================
## Ch 12: THE TWO-NOTRUMP OPENING
## ============================================================
## Weak, 5-5+ in the majors or in the minors (already defined in
## Ch 1). Responses commit to one of opener's (ambiguous) suit pairs;
## since which pair opener actually holds isn't knowable from
## responder's hand alone, the preference bids below are necessarily
## approximate.

$.2N.3C. := Points <= 8 ;
$.2N.3D. := (Diamonds >= 2) AND (Points <= 8) ;
$.2N.3H. := (Hearts >= 2) AND (9 TO 12 Points) ;
$.2N.3S. := (Spades >= 2) AND (9 TO 12 Points) ;
$.2N.3N. := $balanced AND (13 TO 15 Points) ;
$.2N.4C. := $slamish ;
$.2N.4D. := ((Hearts >= 6) AND (Hpts >= 6)) OR ((Spades >= 6) AND (Spts >= 6)) ;
$.2N.4D.4H. := Points >= 0 ;
$.2N.4H. := (Hearts >= 4) AND (Points >= 13) ;
$.2N.4S. := (Spades >= 4) AND (Points >= 13) ;
$.2N.4N. := $slamish ;
$.2N.4N.5C. := $aces0 ;
$.2N.4N.5D. := $aces1 ;
$.2N.4N.5H. := $aces2 ;
$.2N.4N.5S. := $aces3 ;

## ---- Opener's Rebids after 2N-4C ----
$.2N.4C.4D. := (Hearts >= 5 OR Spades >= 5) AND (Points >= 9) ;
$.2N.4C.4H. := (Hearts >= 5 OR Spades >= 5) AND (Points <= 6) ;
$.2N.4C.4S. := (Clubs >= 5 AND Diamonds >= 5) AND (Points >= 9) ;
$.2N.4C.4N. := (Clubs >= 5 AND Diamonds >= 5) AND (Points <= 6) AND (Hearts <= 1) ;
$.2N.4C.5C. := (Clubs >= 5 AND Diamonds >= 5) AND (Points <= 6) AND (Spades <= 1) ;


## ============================================================
## Ch 13: THE TWO-SPADE OPENING
## ============================================================
## Weak, 5-5+ in the reds or in the blacks (already defined in
## Ch 1); the same ambiguous-pair caveat as Ch 12 applies below.

$.2S.2N. := $slamish ;
$.2S.2N.3C. := (Clubs >= 2) AND (Points <= 7) ;
$.2S.2N.3D. := (Diamonds >= 2) AND (Points <= 7) ;
$.2S.2N.3H. := (Hearts >= 2) AND (9 TO 12 Points) ;
$.2S.2N.3S. := (Spades >= 2) AND (9 TO 12 Points) ;
$.2S.3C. := (Clubs >= 2) AND (Points <= 8) ;
$.2S.3D. := ((Diamonds >= 2) OR (Hearts >= 2)) AND (Points <= 8) ;
$.2S.3H. := ((Diamonds >= 2) OR (Hearts >= 2)) AND (Points <= 8) ;
$.2S.3S. := ((Spades >= 2) OR (Clubs >= 2)) AND (Points <= 8) ;
$.2S.3N. := $balanced AND (9 TO 13 Points) ;
$.2S.4C. := $slamish ;
$.2S.4D. := $soundClubSuit OR $soundDiamondSuit OR ((Hearts >= 7) AND (Hpts >= 7)) OR ((Spades >= 7) AND (Spts >= 7)) ;
$.2S.4D.4H. := Points >= 0 ;
$.2S.4H. := (Hearts >= 4) AND ((Clubs >= 4) OR (Diamonds >= 4)) AND (Points >= 9) ;
$.2S.4S. := (Spades >= 4) AND ((Clubs >= 4) OR (Diamonds >= 4)) AND (Points >= 9) ;

## ---- Opener's Rebids after 2S-4C ----
$.2S.4C.4D. := (Hearts >= 5 AND Diamonds >= 5) AND (Points >= 9) ;
$.2S.4C.4H. := (Hearts >= 5 AND Diamonds >= 5) AND (Points <= 6) ;
$.2S.4C.4S. := (Spades >= 5 AND Clubs >= 5) AND (Points <= 6) ;
$.2S.4C.4N. := (Spades >= 5 AND Clubs >= 5) AND (Points >= 9) AND (Diamonds <= 1) ;
$.2S.4C.5C. := (Spades >= 5 AND Clubs >= 5) AND (Points >= 9) AND (Hearts <= 1) ;
$.2S.4C.5D. := (Diamonds <= 1) AND (Points >= 9) AND (Sa OR Ha OR Da OR Ca) ;
$.2S.4C.5H. := (Hearts <= 1) AND (Points >= 9) AND (Sa OR Ha OR Da OR Ca) ;
$.2S.4C.5S. := (Spades <= 1) AND (Points >= 9) AND (Sa OR Ha OR Da OR Ca) ;
$.2S.4C.5N. := (Clubs <= 1) AND (Points >= 9) AND (Sa OR Ha OR Da OR Ca) ;
END
