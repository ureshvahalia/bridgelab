$Any := (Points >= 0);
$ntop := (shape [4,3,3,3] OR [4,4,3,2] OR [5,3,3,2]);
$offshapent := (shape [5,4,2,2] OR [6,3,2,2]) AND (Hearts < 5) AND (Spades < 5);
$.1N. := (15 TO 17 Points) AND (($ntop) OR ($offshapent));
$.1N.P.     := (Points < 8) AND (2 TO 4 Spades) AND (2 TO 4 Hearts) AND (2 TO 5 Diamonds) AND (2 TO 5 Clubs);
$.1N.2D.    := (Hearts > 4) AND (Spades < 5);
$.1N.2H.    := (Spades > 4);
$.1N.4D.	:= (Hearts ?= 6) AND (9 TO 11 Points);
$.1N.4H.	:= (Spades ?= 6) AND (9 TO 11 Points);
$.1N.2C.    := (Points > 7) AND ((Hearts ?= 4) OR (Spades ?= 4)) AND ($ntop or $offshapent);
$.1N.2C.2H.	:= (Hearts > 3);
$.1N.2C.2S.	:= (Hearts < 4) AND (Spades > 3);
$.1N.2C.2D.	:= (Hearts < 4) AND (Spades < 4);
$.1N.2D.2H.	:= (Points > 0);
$.1N.2H.2S.	:= (Points > 0);
$.1N.2D.2H.3N.	:= (10 TO 13 Points) AND (shape [5,3,3,2]);
$.1N.2H.2S.3N.	:= (10 TO 13 Points) AND (shape [5,3,3,2]);
$.1N.2D.2H.4H.	:= (12 TO 13 Points) AND (shape [6,3,3,2]);
$.1N.2H.2S.4S.	:= (12 TO 13 Points) AND (shape [6,3,3,2]);
$balNoMajor	:= $ntop AND (Spades < 4) AND (Hearts < 4);
$.1N.3N.    :=  $balNoMajor AND (10 TO 14 Points);
$.1N.4N.    := $ntop AND (Spades < 4) AND (Hearts < 4) AND (15 TO 16 Points);
$.1N.2N.	:= (Clubs > 4) AND (Diamonds > 4);
$.1N.2S.	:= ($balNoMajor AND (8 TO 9 Points)) OR ((Clubs > 5) AND ((Points < 10) OR ((Points > 13) OR (Spades ?= 4) OR (Hearts ?= 4)))) OR ((Diamonds > 5) AND (Points < 10));
$.1N.2S.2N.	:= (Points ?= 15);
$.1N.2S.3C.	:= (Points > 15);
$.1N.3C.	:= ((Diamonds > 5) AND ((Points < 8) OR ((Points > 13) OR (Spades ?= 4) OR (Hearts ?= 4))));
$.1N.3C.3D.	:= (Points > 0);
$.1N.3C.3D.3H.  := (Hearts ?=4);
$.1N.3C.3D.3S.  := (Spades ?=4);
$.1N.2S.3C.3H.  := (Hearts ?=4);
$.1N.2S.3C.3S.  := (Spades ?=4);
$.1N.2S.2N.3H.  := (Hearts ?=4);
$.1N.2S.2N.3S.  := (Spades ?=4);
$.1N.2S.2N.3C.  := (Points < 8);
$.1N.2S.2N.3C.P.  := (Points > 0);
$.1N.3H.	:= (Points > 9) AND (Hearts < 2) AND (Spades < 5) AND (Diamonds < 7) AND (Clubs < 7);
$.1N.3S.	:= (Points > 9) AND (Spades < 2) AND (Hearts < 5) AND (Diamonds < 7) AND (Clubs < 7);
$notBalanced :=  NOT ($ntop);
$.P. := (Points < 14);
$NotPass    := NOT $.1N.P.
